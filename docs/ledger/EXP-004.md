from math import isclose

def slope(xs, ys, intercept=False):
    if intercept:
        mx, my = sum(xs) / len(xs), sum(ys) / len(ys)
        xs, ys = [x - mx for x in xs], [y - my for y in ys]
    return sum(x * y for x, y in zip(xs, ys)) / sum(x * x for x in xs)

x = [-0.2, -0.1, 0.1, 0.2]
y = [v + 0.3 for v in x]
base = sum(y) / len(y)
overlay = sum(v if p >= 0 else 0 for p, v in zip(x, y)) / len(y)
assert isclose(slope(x, y), 1.0)
assert isclose(slope(x, y, True), 1.0)
assert overlay < base
print(f'A: slopes={slope(x, y):.6f}/{slope(x, y, True):.6f}; '
      f'C1={base:.6f}; overlay={overlay:.6f}; delta={overlay-base:.6f}')

# Forty neighbours have positive association; four eligible names the opposite.
pool_x = x * 44
pool_y = x * 40 + [-v for v in x] * 4
assert isclose(slope(pool_x, pool_y), 36 / 44)
assert isclose(slope(x, [-v for v in x]), -1.0)
print(f'B: pooled slope={slope(pool_x, pool_y):.6f}; eligible slope=-1.000000')

gas_bp = [2 * dollars / 18 * 10000 for dollars in [0.02, 0.035]]
assert isclose(gas_bp[0], 22.22222222222222)
assert isclose(gas_bp[1], 38.88888888888889)
print(f'C: illustrative round-trip gas={gas_bp[0]:.6f}..{gas_bp[1]:.6f} bp')
